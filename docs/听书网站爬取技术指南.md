### 通用爬虫基类

```python
import requests
from bs4 import BeautifulSoup
from fake_useragent import UserAgent
import time
import os

class AudioSpider:
    """听书网站爬虫基类"""
    
    def __init__(self, base_url: str, download_dir: str = "./downloads"):
        self.base_url = base_url
        self.download_dir = download_dir
        self.session = requests.Session()
        self.ua = UserAgent()
        self._setup_session()
        
    def _setup_session(self):
        """配置Session"""
        self.session.headers.update({
            'User-Agent': self.ua.random,
            'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8',
            'Accept-Language': 'zh-CN,zh;q=0.9,en;q=0.8',
            'Connection': 'keep-alive',
        })
    
    def get_page(self, url: str, retry: int = 3) -> BeautifulSoup | None:
        """获取并解析页面"""
        for i in range(retry):
            try:
                resp = self.session.get(url, timeout=10)
                resp.encoding = resp.apparent_encoding
                return BeautifulSoup(resp.text, 'lxml')
            except Exception as e:
                print(f"请求失败 ({i+1}/{retry}): {e}")
                time.sleep(2)
        return None
    
    def download_audio(self, audio_url: str, filename: str) -> bool:
        """下载音频文件"""
        os.makedirs(self.download_dir, exist_ok=True)
        filepath = os.path.join(self.download_dir, filename)
        
        try:
            resp = self.session.get(audio_url, stream=True, timeout=30)
            with open(filepath, 'wb') as f:
                for chunk in resp.iter_content(chunk_size=8192):
                    f.write(chunk)
            return True
        except Exception as e:
            print(f"下载失败: {e}")
            return False
```

## 1️⃣ 书音FM (shuyinfm.com)

### 网站特点
- **类型**: 移动端优化页面
- **内容**: 有声小说、评书、广播剧、儿童故事
- **URL结构**: `/book/{book_id}/` 书籍详情页

### 爬取方案

```python
class ShuyinFMSpider(AudioSpider):
    """书音FM爬虫"""
    
    def __init__(self):
        super().__init__("https://m.shuyinfm.com")
    
    def get_book_list(self, category: str = "xiaoshuo", page: int = 1) -> list:
        """获取书籍列表
        分类: xiaoshuo(小说), pingshu(评书), guangboju(广播剧), ertong(儿童)
        """
        url = f"{self.base_url}/{category}/list_{page}.html"
        soup = self.get_page(url)
        if not soup:
            return []
        
        books = []
        for item in soup.select('.book-item'):  # 根据实际结构调整选择器
            book = {
                'title': item.select_one('.title').text.strip(),
                'url': self.base_url + item.select_one('a')['href'],
                'cover': item.select_one('img')['src'],
            }
            books.append(book)
        return books
    
    def get_audio_list(self, book_url: str) -> list:
        """获取书籍的音频列表"""
        soup = self.get_page(book_url)
        if not soup:
            return []
        
        audios = []
        for item in soup.select('.chapter-list a'):  # 根据实际结构调整
            audios.append({
                'title': item.text.strip(),
                'url': self.base_url + item['href'],
            })
        return audios

# 使用示例
spider = ShuyinFMSpider()
books = spider.get_book_list("xiaoshuo", page=1)
for book in books[:3]:
    print(f"书籍: {book['title']}")
```

## 2️⃣ 七八听书网 (ting78.com)

### 网站特点
- **类型**: 传统静态网站
- **内容**: 玄幻、言情、都市、恐怖、武侠等
- **URL结构**: `/mp3/{book_id}.html` 书籍页面

### 爬取方案

```python
class Ting78Spider(AudioSpider):
    """七八听书网爬虫"""
    
    def __init__(self):
        super().__init__("https://www.ting78.com")
    
    def get_category_books(self, category_id: int = 1, page: int = 1) -> list:
        """获取分类书籍
        分类ID: 1-玄幻, 2-言情, 3-都市, 4-恐怖, 5-惊悚, 6-推理, 7-武侠
        """
        url = f"{self.base_url}/books/{category_id}_{page}.html"
        soup = self.get_page(url)
        if not soup:
            return []
        
        books = []
        for item in soup.select('.listbox li'):
            a_tag = item.select_one('a')
            if a_tag:
                books.append({
                    'title': a_tag.text.strip(),
                    'url': self.base_url + a_tag['href'],
                })
        return books
    
    def get_audio_urls(self, book_url: str) -> list:
        """解析书籍页面获取音频列表"""
        soup = self.get_page(book_url)
        if not soup:
            return []
        
        # 音频通常在playlist或JavaScript中
        # 需要分析具体的音频地址模式
        audios = []
        script_tags = soup.find_all('script')
        for script in script_tags:
            if script.string and 'mp3' in script.string:
                # 使用正则提取音频URL
                import re
                urls = re.findall(r'https?://[^\s"\']+\.mp3', script.string)
                audios.extend(urls)
        
        return list(set(audios))

# 使用示例
spider = Ting78Spider()
books = spider.get_category_books(1, 1)  # 玄幻分类第一页
```

## 3️⃣ 听书迷 (tingsm.com)

### 网站特点
- **类型**: 传统听书站
- **内容**: 有声小说、鬼故事、广播剧
- **URL结构**: `/book/{book_id}.html`

### 爬取方案

```python
class TingSMSpider(AudioSpider):
    """听书迷爬虫"""
    
    def __init__(self):
        super().__init__("https://www.tingsm.com")
    
    def get_latest_books(self) -> list:
        """获取最新连载"""
        url = f"{self.base_url}/Serial.html"
        soup = self.get_page(url)
        if not soup:
            return []
        
        books = []
        for item in soup.select('.book-list a'):
            books.append({
                'title': item.text.strip(),
                'url': self.base_url + item['href'],
            })
        return books
    
    def parse_book_page(self, book_url: str) -> dict:
        """解析书籍详情页"""
        soup = self.get_page(book_url)
        if not soup:
            return {}
        
        return {
            'title': soup.select_one('h1').text.strip() if soup.select_one('h1') else '',
            'author': soup.select_one('.author').text.strip() if soup.select_one('.author') else '',
            'chapters': [
                {'title': a.text, 'url': a['href']}
                for a in soup.select('.chapter-list a')
            ]
        }
```

## 4️⃣ 乐听吧 (leting8.com)

### 网站特点
- **类型**: 综合性有声站（与ting78结构相似）
- **内容**: 玄幻、言情、恐怖、评书、戏曲等
- **URL结构**: `/so/{book_id}.html`

### 爬取方案

```python
class LeTing8Spider(AudioSpider):
    """乐听吧爬虫"""
    
    def __init__(self):
        super().__init__("https://www.leting8.com")
    
    def search(self, keyword: str) -> list:
        """搜索书籍"""
        url = f"{self.base_url}/search.php"
        params = {'searchword': keyword}
        
        try:
            resp = self.session.get(url, params=params, timeout=10)
            resp.encoding = 'utf-8'
            soup = BeautifulSoup(resp.text, 'lxml')
            
            results = []
            for item in soup.select('.search-result li'):
                a_tag = item.select_one('a')
                if a_tag:
                    results.append({
                        'title': a_tag.text.strip(),
                        'url': self.base_url + a_tag['href'],
                    })
            return results
        except Exception as e:
            print(f"搜索失败: {e}")
            return []
    
    def get_category_list(self, category_id: int) -> list:
        """获取分类列表
        分类: 1-玄幻, 2-言情, 3-都市, 4-恐怖, 5-惊悚, 6-推理
        """
        url = f"{self.base_url}/le/{category_id}.html"
        soup = self.get_page(url)
        if not soup:
            return []
        
        books = []
        for a in soup.select('.list a[href*="/so/"]'):
            books.append({
                'title': a.text.strip(),
                'url': self.base_url + a['href'],
            })
        return books
```

## 5️⃣ 猫耳FM (missevan.com)

### 网站特点
- **类型**: 商业化二次元音频平台
- **内容**: 广播剧、ASMR、翻唱、有声小说
- **特点**: 有完整API系统，需要分析接口

### 爬取方案

> ⚠️ **注意**: 猫耳FM是商业平台，部分内容需要登录或付费，请遵守平台规则

```python
import re
import json

class MissevanSpider(AudioSpider):
    """猫耳FM爬虫"""
    
    def __init__(self):
        super().__init__("https://www.missevan.com")
        self.api_base = "https://www.missevan.com/sound"
    
    def get_sound_info(self, sound_id: int) -> dict:
        """获取音频信息
        
        Args:
            sound_id: 音频ID，可从URL中获取，如 /sound/player?id=123456
        """
        url = f"{self.api_base}/getsound?soundid={sound_id}"
        
        try:
            resp = self.session.get(url, timeout=10)
            data = resp.json()
            
            if data.get('success'):
                info = data.get('info', {}).get('sound', {})
                return {
                    'id': info.get('id'),
                    'title': info.get('soundstr'),
                    'duration': info.get('duration'),
                    'audio_url': info.get('soundurl'),  # MP3地址
                    'cover': info.get('front_cover'),
                    'username': info.get('username'),
                }
            return {}
        except Exception as e:
            print(f"获取音频信息失败: {e}")
            return {}
    
    def get_drama_episodes(self, drama_id: int) -> list:
        """获取广播剧集列表"""
        url = f"https://www.missevan.com/dramaapi/getdrama?drama_id={drama_id}"
        
        try:
            resp = self.session.get(url, timeout=10)
            data = resp.json()
            
            if data.get('success'):
                episodes = data.get('info', {}).get('episodes', {}).get('episode', [])
                return [
                    {
                        'id': ep.get('sound_id'),
                        'title': ep.get('soundstr'),
                        'duration': ep.get('duration'),
                    }
                    for ep in episodes
                ]
            return []
        except Exception as e:
            print(f"获取剧集列表失败: {e}")
            return []
    
    def search_sounds(self, keyword: str, page: int = 1) -> list:
        """搜索音频"""
        url = "https://www.missevan.com/sound/search"
        params = {
            's': keyword,
            'p': page,
        }
        
        try:
            resp = self.session.get(url, params=params, timeout=10)
            # 解析搜索结果页面
            soup = BeautifulSoup(resp.text, 'lxml')
            
            results = []
            for item in soup.select('.sound-item'):
                sound_id = item.get('data-id')
                if sound_id:
                    results.append({
                        'id': int(sound_id),
                        'title': item.select_one('.title').text.strip(),
                    })
            return results
        except Exception as e:
            print(f"搜索失败: {e}")
            return []

# 使用示例
spider = MissevanSpider()
info = spider.get_sound_info(123456)  # 替换为实际sound_id
if info.get('audio_url'):
    print(f"音频地址: {info['audio_url']}")
```

### 1. 异步批量下载

```python
import aiohttp
import asyncio
from pathlib import Path

async def download_audio_async(session: aiohttp.ClientSession, url: str, filepath: str):
    """异步下载单个音频"""
    try:
        async with session.get(url) as resp:
            if resp.status == 200:
                Path(filepath).parent.mkdir(parents=True, exist_ok=True)
                with open(filepath, 'wb') as f:
                    async for chunk in resp.content.iter_chunked(8192):
                        f.write(chunk)
                return True
    except Exception as e:
        print(f"下载失败 {url}: {e}")
    return False

async def batch_download(audio_list: list, download_dir: str, max_concurrent: int = 5):
    """批量异步下载"""
    semaphore = asyncio.Semaphore(max_concurrent)
    
    async def limited_download(session, url, filepath):
        async with semaphore:
            return await download_audio_async(session, url, filepath)
    
    async with aiohttp.ClientSession() as session:
        tasks = [
            limited_download(session, audio['url'], f"{download_dir}/{audio['filename']}")
            for audio in audio_list
        ]
        results = await asyncio.gather(*tasks)
        return results

# 使用示例
# audio_list = [{'url': 'http://...', 'filename': '01.mp3'}, ...]

```

### 2. 代理池支持

```python
class ProxyPool:
    """简单代理池"""
    
    def __init__(self, proxies: list):
        self.proxies = proxies
        self.index = 0
    
    def get_proxy(self) -> dict:
        if not self.proxies:
            return {}
        proxy = self.proxies[self.index % len(self.proxies)]
        self.index += 1
        return {'http': proxy, 'https': proxy}

# 使用示例
pool = ProxyPool(['http://127.0.0.1:7890', 'http://127.0.0.1:7891'])
response = requests.get(url, proxies=pool.get_proxy())
```

### 3. 断点续传

```python
def download_with_resume(url: str, filepath: str, session: requests.Session):
    """支持断点续传的下载"""
    headers = {}
    
    # 检查已下载的部分
    if os.path.exists(filepath):
        downloaded_size = os.path.getsize(filepath)
        headers['Range'] = f'bytes={downloaded_size}-'
        mode = 'ab'  # 追加模式
    else:
        mode = 'wb'
    
    resp = session.get(url, headers=headers, stream=True)
    
    with open(filepath, mode) as f:
        for chunk in resp.iter_content(chunk_size=8192):
            f.write(chunk)
```

## ⚠️ 注意事项

### 法律合规
1. 仅供个人学习研究使用
2. 遵守各网站的 `robots.txt` 规则
3. 不要将爬取的内容用于商业用途
4. 尊重版权，支持正版

### 技术规范
1. 控制请求频率，建议间隔 1-3 秒
2. 设置合理的超时时间
3. 处理好异常情况，避免程序崩溃
4. 使用随机 User-Agent

### 反爬应对
1. 使用代理IP池
2. 模拟浏览器请求头
3. 处理验证码（如有）
4. 使用 Selenium/Playwright 处理动态加载
