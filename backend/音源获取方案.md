## 项目概述

### 项目背景
基于对以下两个开源项目的深度分析：
- **Ximalaya-Downloader**: 喜马拉雅音频破解工具，采用逆向工程技术
- **tingshu**: 聚合听书客户端，支持30+听书网站API

### 核心价值
采用**API聚合**而非**逆向破解**的技术路线，确保：
- ✅ 合法合规 (调用公开API)
- ✅ 稳定可靠 (避免加密算法变更风险)
- ✅ 易于维护 (模块化设计)
- ✅ 快速扩展 (标准化接口)

### 策略二：网页数据解析（备选）

#### 适用场景
当目标网站没有公开API时，使用HTML解析方式。

#### 实现方式
```kotlin
import org.jsoup.Jsoup

fun parseHtmlPage(url: String): List<Book> {
    val doc = Jsoup.connect(url)
        .header("User-Agent", "Mozilla/5.0...")
        .timeout(30000)
        .get()

    return doc.select(".book-item").map { element ->
        Book(
            coverUrl = element.select("img.cover").attr("src"),
            bookUrl = element.select("a.book-link").attr("href"),
            title = element.select("h3.title").text(),
            author = element.select(".author").text(),
            artist = element.select(".artist").text(),
            status = element.select(".status").text(),
            intro = element.select(".intro").text(),
            sourceId = "source_id"
        )
    }
}
```

#### 优点
- 适用范围广
- 不需要破解加密
- 数据实时性强

#### 缺点
- 网页结构变化需及时维护
- 请求频率受限
- 解析性能较低

### 源适配实施

#### 酷我源（快速入门）

1. **创建源文件**
```kotlin
package com.yourproject.sources

import com.yourproject.model.*
import com.github.kittinunf.fuel.Fuel
import com.github.kittinunf.fuel.json.responseJson

class KuWoSource : AudioSource {
    override fun getSourceId() = "kuwo_001"
    override fun getName() = "酷我畅听"
    override fun getUrl() = "http://kuwo.cn"

    override fun search(keywords: String, page: Int): Pair<List<Book>, Int> {
        val encodedKeywords = URLEncoder.encode(keywords, "utf-8")
        val url = "http://baby.kuwo.cn/tingshu/api/search/Search" +
                 "?rn=10&type=album&pn=$page&wd=$encodedKeywords"

        val result = Fuel.get(url).responseJson()

        // 解析JSON并转换为Book列表
        val data = result.third.get().obj().getJSONObject("data")
        val books = data.getJSONArray("data").let { array ->
            (0 until array.length()).map { index ->
                val item = array.getJSONObject(index)
                Book(
                    coverUrl = item.getString("coverImg"),
                    bookUrl = "http://baby.kuwo.cn/tingshu/api/data/album/songs?albumId=${item.getInt("albumId")}",
                    title = item.getString("albumName"),
                    author = "",
                    artist = item.getString("artistName"),
                    status = "共 ${item.getInt("songTotal")} 章",
                    intro = item.getString("title"),
                    sourceId = getSourceId()
                )
            }
        }

        val total = data.getInt("total")
        val totalPages = ceil(total / 10.0).toInt()

        return Pair(books, totalPages)
    }

    override fun getBookDetail(bookUrl: String, loadEpisodes: Boolean): BookDetail {
        val episodes = arrayListOf<Episode>()

        if (loadEpisodes) {
            val data = Fuel.get(bookUrl).responseJson().third.get()
                .obj().getJSONArray("data")

            (0 until data.length()).forEach { index ->
                val item = data.getJSONObject(index)
                episodes.add(
                    Episode(
                        name = item.getString("name"),
                        url = "http://antiserver.kuwo.cn/anti.s?useless=/resource/&format=mp3" +
                             "&rid=MUSIC_${item.getString("musicrid")}&response=res&type=convert_url"
                    )
                )
            }
        }

        return BookDetail(episodes)
    }

    override fun getAudioUrl(episodeUrl: String): String {
        // 解析真实的音频地址
        return Fuel.get(episodeUrl).responseString().third.get()
    }
}
```

#### 懒人听书源（进阶示例）

```kotlin
class LanRenSource : AudioSource {
    private val baseUrl = "https://m.lrts.me"

    override fun getSourceId() = "lanren_001"
    override fun getName() = "懒人听书"

    override fun search(keywords: String, page: Int): Pair<List<Book>, Int> {
        val encodedKeywords = URLEncoder.encode(keywords, "utf-8")
        val url = "${baseUrl}/ajax/search" +
                 "?keyWord=$encodedKeywords" +
                 "&pageSize=40&pageNum=$page&searchOption=1"

        val result = Fuel.get(url).header("User-Agent", getMobileUA())
            .responseJson().third.get().obj()

        val bookResult = result.getJSONObject("data").getJSONObject("bookResult")
        val totalPage = ceil(bookResult.getInt("count") / 40.0).toInt()

        val books = bookResult.getJSONArray("list").let { array ->
            (0 until array.length()).map { index ->
                val book = array.getJSONObject(index)
                Book(
                    coverUrl = book.getString("cover"),
                    bookUrl = "$baseUrl?${book.getLong("id")}",
                    title = book.getString("name"),
                    author = book.getString("author"),
                    artist = book.getString("announcer"),
                    status = "连载中",
                    intro = book.optString("recReason", book.getString("desc")),
                    sourceId = getSourceId(),
                    isFree = !book.getJSONArray("tags").toString().contains("VIP")
                )
            }
        }

        return Pair(books, totalPage)
    }

    override fun getBookDetail(bookUrl: String, loadEpisodes: Boolean): BookDetail {
        val bookId = bookUrl.substringAfter("?")
        val episodes = mutableListOf<Episode>()

        if (loadEpisodes) {
            val pageSize = 200
            val url = "${baseUrl}/ajax/getBookMenu" +
                     "?bookId=$bookId&pageNum=1&pageSize=$pageSize&sortType=0"

            val result = Fuel.get(url).header("User-Agent", getMobileUA())
                .responseJson().third.get().obj()

            val tracks = result.getJSONArray("list")
            episodes.addAll(getEpisodes(tracks, bookId, 0))
        }

        return BookDetail(episodes)
    }

    private fun getEpisodes(tracks: JSONArray, bookId: String, offset: Int): List<Episode> {
        return (0 until tracks.length()).map { index ->
            val track = tracks.getJSONObject(index)
            val section = track.getInt("section")
            val isFree = track.getInt("payType") == 0
            val episodeUrl = if (isFree) {
                "${baseUrl}/ajax/getPlayPath?entityId=$bookId" +
                "&entityType=3&opType=1&sections=[$section]&type=0"
            } else {
                "${baseUrl}/player?index=$index&entityType=3" +
                "&sonId=${track.getLong("id")}&id=$bookId"
            }

            Episode(
                name = track.getString("name"),
                url = episodeUrl,
                isFree = isFree
            )
        }
    }

    override fun getAudioUrl(episodeUrl: String): String {
        return if (episodeUrl.contains("getPlayPath")) {
            // 免费音频直接获取
            val data = Fuel.get(episodeUrl).responseJson().third.get()
                .obj().getJSONArray("list").getJSONObject(0)
            data.getString("path")
        } else {
            // VIP音频需要WebView嗅探（简化版）
            // 实际应使用Headless浏览器或WebView组件
            throw Exception("VIP内容需登录后播放")
        }
    }
}
```

### 错误处理与重试机制

```kotlin
suspend fun <T> retryHttp(
    maxRetries: Int = 3,
    delayMillis: Long = 1000,
    block: suspend () -> T
): Result<T> {
    repeat(maxRetries - 1) { attempt ->
        try {
            return Result.success(block())
        } catch (e: Exception) {
            if (e is CancellationException) throw e
            delay(delayMillis * (attempt + 1))
        }
    }
    return try {
        Result.success(block())
    } catch (e: Exception) {
        Result.failure(e)
    }
}

// 使用示例
suspend fun searchWithRetry(source: AudioSource, keywords: String) {
    val result = retryHttp {
        source.search(keywords, 1)
    }

    result.onSuccess { (books, total) ->
        // 处理成功结果
    }.onFailure { exception ->
        // 记录日志，提示用户
        logger.error("搜索失败: ${exception.message}", exception)
    }
}
```

   - 避免逆向破解加密
   - 遵守robots.txt

2. **版权声明**

   - 遵守各平台服务条款
   - 考虑增加"仅供个人学习"声明

   ```kotlin
   // 限制请求频率
   val rateLimiter = RateLimiter.create(10.0) // 每秒10个请求

   fun fetchWithRateLimit(url: String) {
       rateLimiter.acquire()
   ```
